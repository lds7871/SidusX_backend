#user  nobody;
worker_processes  1;

#error_log  logs/error.log;
#pid        logs/nginx.pid;

events {
    worker_connections  1024;
}

http {
    include       mime.types;
    default_type  application/octet-stream;

    access_log  logs/access.log;

    sendfile        on;
    keepalive_timeout  65;

    # 定义后端服务地址
    upstream app_backend {
        server 127.0.0.1:8090;  # X自动，QQ机器人 服务监听端口
        server 127.0.0.1:8100;  # 毕业设计 服务监听端口
    }

    # 定义 Vue 前端服务地址
    upstream vue_app {
        server 127.0.0.1:5173;  # Vue Vite 开发服务或生产预览服务
    }

    # HTTP 入口（用于 Twitter OAuth 回调）
    server {
        listen       80;
        server_name  115.190.170.56;

        location / {
            proxy_pass http://app_backend;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            proxy_http_version 1.1;
            proxy_read_timeout 90s;
            proxy_connect_timeout 10s;
            proxy_send_timeout 10s;
        }

        # 可选：强制跳转到 HTTPS
        # return 301 https://$host$request_uri;
    }

    # Vue 前端应用（HTTP）- 重定向到 HTTPS
    server {
        listen       80;
        server_name  zh-company.top www.zh-company.top;

        return 301 https://$server_name$request_uri;
    }

    # HTTPS 入口（推荐生产环境使用）
    server {
        listen 443 ssl;
        http2 on;
        server_name 115.190.170.56;

        # SSL 证书路径（请确保文件存在）
        ssl_certificate     C:/Users/Administrator/Desktop/ServerSync/Nginx-1.28.0/ssl/fullchain.pem;
        ssl_certificate_key C:/Users/Administrator/Desktop/ServerSync/Nginx-1.28.0/ssl/privkey.pem;

        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_prefer_server_ciphers on;
        ssl_ciphers "ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384:ECDHE-ECDSA-AES128-GCM-SHA256";
        add_header Strict-Transport-Security "max-age=63072000; includeSubDomains; preload" always;

       # 阻止访问隐藏文件和目录
        location ~ /\. {
            deny all;
            access_log off;
            log_not_found off;
        }

        # 阻止访问敏感配置文件
        location ~ \.(git|svn|hg) {
            deny all;
            access_log off;
            log_not_found off;
        }

        # 阻止访问 IDE 配置文件
        location ~ /\.vscode/ {
            deny all;
            access_log off;
            log_not_found off;
        }

        location ~ /\.idea/ {
            deny all;
            access_log off;
            log_not_found off;
        }

        # 仅允许以 /api 或 /GHapi 开头的请求代理到后端
        location ~ ^/(api|GHapi) {
            proxy_pass http://app_backend;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;

            # WebSocket 支持
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";

            proxy_http_version 1.1;
            proxy_read_timeout 90s;
            proxy_connect_timeout 10s;
            proxy_send_timeout 10s;
        }

        # 拒绝其他所有请求
        location / {
            return 403 "弱智";
        }

        # 健康检查接口（Spring Boot Actuator）
        location /actuator/health {
            proxy_pass http://app_backend/actuator/health;
            proxy_set_header X-Forwarded-Host $host;
        }

        # 静态资源缓存优化
        location ~* \.(js|css|png|jpg|jpeg|gif|svg|ico)$ {
            proxy_pass http://app_backend;
            expires 7d;
            add_header Cache-Control "public";
        }
    }

    # Vue 前端应用（HTTPS） - 静态文件服务
    server {
        listen 443 ssl;
        http2 on;
        server_name zh-company.top www.zh-company.top;

        # SSL 证书路径（使用现有证书，请确保证书支持 zh-company.top 域名）
        ssl_certificate     C:/Users/Administrator/Desktop/ServerSync/Nginx-1.28.0/ssl/zh-company.top.pem;
        ssl_certificate_key C:/Users/Administrator/Desktop/ServerSync/Nginx-1.28.0/ssl/zh-company.top.key;

        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_prefer_server_ciphers on;
        ssl_ciphers "ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384:ECDHE-ECDSA-AES128-GCM-SHA256";
        add_header Strict-Transport-Security "max-age=63072000; includeSubDomains; preload" always;

        root C:\Users\Administrator\Desktop\ZHcompany\dist;
        index index.html;

        # API 代理到后端服务
        location /api/ {
            proxy_pass http://app_backend;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            proxy_http_version 1.1;
            proxy_read_timeout 90s;
            proxy_connect_timeout 10s;
            proxy_send_timeout 10s;
        }

        # Vue Router 配置 - 所有路由都返回 index.html
        location / {
            try_files $uri $uri/ /index.html;
        }

        # 静态资源缓存优化
        location ~* \.(js|css|png|jpg|jpeg|gif|svg|ico|woff|woff2)$ {
            expires 30d;
            add_header Cache-Control "public, immutable";
        }
    }
}
